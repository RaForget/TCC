// --------------------- Comandos Simulação ----------------------

Para rodar a simulaçao do robô

ros2 launch trekking_2025 simulation.launch.py

// ---------------------------------------------------------------

Para visualização dos dados e do robô

ros2 run rviz2 rviz2

// ---------------------------------------------------------------

Para rodar o SLAM (modo localização - mapa previamente gerado)

ros2 launch trekking_2025 slam.launch.localization.py

// ---------------------------------------------------------------

Para ativar o modo de navegação autônoma do robô (NAV2)

ros2 launch trekking_2025 navigation_launch.py use_sim_time:=true
